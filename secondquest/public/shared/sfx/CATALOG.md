## Use

```python
from sfx_pick import Picker            # scripts/sfx_pick.py
pk = Picker(seed='ep004')            # one per video
path = pk.file('whoosh', t=12.3)      # varied render (cached wav), never the same file twice within 20 s
y = pk.layer('whoosh+foley-cloth', t=14.0)   # numpy stereo, layered
```
